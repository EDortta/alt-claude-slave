def summarize_events(events):
    result = {}
    for event in events:
        if 'user' not in event or 'action' not in event or 'value' not in event:
            raise ValueError("Event must have user, action, and value")
        if event['user'] not in result:
            result[event['user']] = {
                'count': 1,
                'total': event['value'],
                'actions': [event['action']]
            }
        else:
            result[event['user']]['count'] += 1
            result[event['user']]['total'] += event['value']
            result[event['user']]['actions'].append(event['action'])
    return result
