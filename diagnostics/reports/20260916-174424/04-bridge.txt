  -o LogLevel=ERROR \
  -o BatchMode=yes \
  -o ConnectTimeout=10 \
  -o ServerAliveInterval=30 \
  -o ServerAliveCountMax=3 \
  'esteban@dom1.inovacaosistemas.com.br' \
  "sudo -n incus exec 'alt-claude-slave' \
    --cwd /srv/alt-claude/repos/alt-claude-slave \
    --env HOME=/home/slave \
    --env LLAMA_CACHE=/srv/alt-claude/models \
    --env LLAMA_BIN=/home/slave/.local/opt/llama.cpp/bin \
    -- runuser -u slave -- python3 '/srv/alt-claude/repos/alt-claude-slave/scripts/slave_mcp.py'" \
  2> >(tee -a "$LOG_DIR/mcp-stderr.log" >&2)
